func hashBytes(_ bytes: Data) -> String { SHA256.hash(data: bytes).map { String(format: "%02x", $0) }.joined() }
func shaped(_ text: String, version: UInt8) -> String {
    var b = Array(SHA256.hash(data: Data(text.utf8)).prefix(16)); b[6] = (b[6] & 15) | version; b[8] = (b[8] & 63) | 128
    let h = b.map { String(format: "%02x", $0) }.joined()
    return "\(h.prefix(8))-\(h.dropFirst(8).prefix(4))-\(h.dropFirst(12).prefix(4))-\(h.dropFirst(16).prefix(4))-\(h.dropFirst(20))"
}
func uuid(_ t: String) -> String { shaped(t, version: 0x50) }
func operationId(_ t: String) -> String { shaped(t, version: 0x40) }
func canonical<T: Encodable>(_ v: T) -> Data { let e = JSONEncoder(); e.outputFormatting = [.sortedKeys, .withoutEscapingSlashes]; return try! e.encode(v) }
let store = "2b7e1516-28ae-4d2a-a6d2-abf7158809cf", device = "e3c75f30-363e-4323-a592-cf07fe14c4a6"
func cell(_ t: String, _ v: String?) -> CloudBrainMemoryCell { .init(type: t, value: v) }
var out: [String: String] = [:]
func row(_ name: String, _ table: String, _ columns: [String], _ cells: [CloudBrainMemoryCell], keys: [String], refs: [String: CloudBrainMemoryReference] = [:], rowId: String, stamp: Int64) {
    let body = canonical(CloudBrainMemoryRow(table: table, columns: columns, cells: cells, references: refs))
    let keyCells = keys.map { cells[columns.firstIndex(of: $0)!] }
    let key = "row/" + table + "/" + hashBytes(canonical(keyCells))
    let hash = hashBytes(body)
    let version = uuid(store + ":" + key + "\n" + hash)
    let part = CloudBrainMemoryPart(storeId: store, inventoryRow: rowId, key: key, version: version, resourceHash: hash, bytes: String(body.count), index: 0, count: 1, parents: [], data: body.base64EncodedString(), stamp: String(stamp), baseline: false)
    let payload = canonical(part)
    out[name + ".body"] = String(decoding: body, as: UTF8.self)
    out[name + ".key"] = key; out[name + ".version"] = version
    out[name + ".payload"] = String(decoding: payload, as: UTF8.self)
    out[name + ".payloadHash"] = hashBytes(payload)
    out[name + ".opId"] = operationId(version + "\n0\n" + device)
    out[name + ".entity"] = uuid(store + "\n" + key + "\n0")
}
